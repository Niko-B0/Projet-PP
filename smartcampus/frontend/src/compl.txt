void clearTrace() { traceCount = 0; }
void addTracePoint(float x, float y) {
  if (traceCount == 0) { traceX[0] = x; traceY[0] = y; traceCount = 1; return; }
  float dx = x - traceX[traceCount-1], dy = y - traceY[traceCount-1];
  if (sqrt(dx*dx + dy*dy) < 0.0005f) return;
  if (traceCount < TRACE_MAX) { traceX[traceCount] = x; traceY[traceCount] = y; traceCount++; }
}
void clearTraceEncodeurs() { traceEncCount = 0; }
void addTracePointEncodeurs(float x, float y) {
  if (traceEncCount == 0) { traceEncX[0] = x; traceEncY[0] = y; traceEncCount = 1; return; }
  float dx = x - traceEncX[traceEncCount-1], dy = y - traceEncY[traceEncCount-1];
  if (sqrt(dx*dx + dy*dy) < 0.1f) return;
  if (traceEncCount < TRACE_ENC_MAX) { traceEncX[traceEncCount] = x; traceEncY[traceEncCount] = y; traceEncCount++; }
}

// =====================================================
// INTERRUPTIONS ENCODEURS

void IRAM_ATTR isrEncG() { countG++; }
void IRAM_ATTR isrEncD() { countD++; }

// =====================================================
// ENCODEURS — utilitaires
// =====================================================
void resetEncodeurs() {
  noInterrupts(); countG = 0; countD = 0; interrupts();
  lastCountG = 0; lastCountD = 0;
}
void lireEncodeurs(long &g, long &d) {
  noInterrupts(); g = countG; d = countD; interrupts();
}
float ticksToCm(long t) { return ((float)t / VARIATIONS_PAR_TOUR) * CIRCONFERENCE_CM; }
long  cmToTicks(float cm) { return (long)((cm / CIRCONFERENCE_CM) * VARIATIONS_PAR_TOUR); }

float normaliserAngle(float deg) {
  while (deg < 0.0f)    deg += 360.0f;
  while (deg >= 360.0f) deg -= 360.0f;
  return deg;
}
float normaliserAngle180(float deg) {
  while (deg >  180.0f) deg -= 360.0f;
  while (deg < -180.0f) deg += 360.0f;
  return deg;
}

void resetOdometryEncodeurs() {
  posX_enc_cm = 0.0f; posY_enc_cm = 0.0f; theta_enc_deg = 0.0f;
  long g, d; lireEncodeurs(g, d);
  lastCountG = g; lastCountD = d;
  clearTraceEncodeurs(); addTracePointEncodeurs(0, 0);
}

void updateOdometryEncodeurs() {
  long g, d; lireEncodeurs(g, d);
  long dg = g - lastCountG, dd = d - lastCountD;
  lastCountG = g; lastCountD = d;
  if (dg == 0 && dd == 0) return;
  float dG = ticksToCm(dg), dD = ticksToCm(dd), dC = (dG + dD) / 2.0f;
  if (etatActuel == AVANT) {
    float r = theta_enc_deg * PI / 180.0f;
    posX_enc_cm += dC * cos(r); posY_enc_cm += dC * sin(r);
    addTracePointEncodeurs(posX_enc_cm, posY_enc_cm);
  } else if (etatActuel == ARRIERE) {
    float r = theta_enc_deg * PI / 180.0f;
    posX_enc_cm -= dC * cos(r); posY_enc_cm -= dC * sin(r);
    addTracePointEncodeurs(posX_enc_cm, posY_enc_cm);
  } else if (etatActuel == GAUCHE) {
    float dm = (dg + dd) / 2.0f;
    theta_enc_deg = normaliserAngle(theta_enc_deg + dm * DEG_PAR_VARIATION);
  } else if (etatActuel == DROITE) {
    float dm = (dg + dd) / 2.0f;
    theta_enc_deg = normaliserAngle(theta_enc_deg - dm * DEG_PAR_VARIATION);
  }
}

void writeRegister(uint8_t addr, uint8_t reg, uint8_t value) {
  Wire.beginTransmission(addr); Wire.write(reg); Wire.write(value); Wire.endTransmission();
}
int16_t read16(uint8_t addr, uint8_t regLow) {
  Wire.beginTransmission(addr); Wire.write(regLow); Wire.endTransmission(false);
  Wire.requestFrom((uint8_t)addr, (uint8_t)2);
  if (Wire.available() < 2) return 0;
  uint8_t lo = Wire.read(), hi = Wire.read();
  return (int16_t)((hi << 8) | lo);
}

// =====================================================
// INIT CAPTEURS
// =====================================================
void initIMU() {
  writeRegister(ADDR_IMU, CTRL3_C,  0b01000100);
  // Accéléromètre 104 Hz, ±2g
  writeRegister(ADDR_IMU, CTRL1_XL, 0b01000000);
  // Gyroscope 104 Hz, ±245 dps  (bits [7:4]=0100 → 104Hz, bits [3:2]=00 → 245dps)
  writeRegister(ADDR_IMU, CTRL2_G,  0b01000000);
}
void initMagnetometre() {
  writeRegister(ADDR_MAG, MAG_CTRL_REG1, 0b01110000);
  writeRegister(ADDR_MAG, MAG_CTRL_REG2, 0b00000000);
  writeRegister(ADDR_MAG, MAG_CTRL_REG3, 0b00000000);
  writeRegister(ADDR_MAG, MAG_CTRL_REG4, 0b00001100);
}

// =====================================================
// CALIBRATION GYROSCOPE
// Mesure l'offset Z au repos (biais de dérive)
// =====================================================
void calibrerGyro() {
  const int N = 500;
  long somme = 0;
  Serial.println("Calibration gyroscope Z...");
  for (int i = 0; i < N; i++) {
    somme += read16(ADDR_IMU, OUTZ_L_G);
    delay(5);
  }
  gyroOffsetZ = ((float)somme / N) * GYRO_SENS_DPS;
  Serial.print("gyroOffsetZ = "); Serial.println(gyroOffsetZ, 6);
}

// =====================================================
// LIRE VITESSE ANGULAIRE Z (degrés/seconde)
// =====================================================
float lireGyroZ_dps() {
  int16_t raw = read16(ADDR_IMU, OUTZ_L_G);
  return raw * GYRO_SENS_DPS - gyroOffsetZ;
}

// =====================================================
// CALIBRATION ACCEL
// =====================================================
void calibrerAccel() {
  const int N = 500;
  long sx = 0, sy = 0;
  Serial.println("Calibration accel...");
  for (int i = 0; i < N; i++) {
    sx += read16(ADDR_IMU, OUTX_L_XL);
    sy += read16(ADDR_IMU, OUTY_L_XL);
    delay(5);
  }
  accelOffsetX = ((float)sx / N) * 0.000061f * 9.81f;
  accelOffsetY = ((float)sy / N) * 0.000061f * 9.81f;
}

void recalibrerAccelInstant() {
  const int N = 50;
  long sx = 0, sy = 0;
  for (int i = 0; i < N; i++) {
    sx += read16(ADDR_IMU, OUTX_L_XL);
    sy += read16(ADDR_IMU, OUTY_L_XL);
    delay(5);
  }
  accelOffsetX = ((float)sx / N) * 0.000061f * 9.81f;
  accelOffsetY = ((float)sy / N) * 0.000061f * 9.81f;
  accelX_ms2 = accelY_ms2 = accelGlobale_ms2 = 0.0f;
  vitesseX_ms = vitesseY_ms = vitesseGlobale_ms = 0.0f;
  immobileCounter = 0; lastUpdateMicros = micros();
}

// =====================================================
// MAGNETOMETRE
// =====================================================
bool lireMagRaw(int16_t &mx, int16_t &my, int16_t &mz) {
  Wire.beginTransmission(ADDR_MAG); Wire.write(MAG_OUT_X_L | 0x80);
  if (Wire.endTransmission(false) != 0) return false;
  Wire.requestFrom((uint8_t)ADDR_MAG, (uint8_t)6);
  if (Wire.available() < 6) return false;
  uint8_t xl=Wire.read(),xh=Wire.read(),yl=Wire.read(),yh=Wire.read(),zl=Wire.read(),zh=Wire.read();
  mx=(int16_t)((xh<<8)|xl); my=(int16_t)((yh<<8)|yl); mz=(int16_t)((zh<<8)|zl);
  return true;
}
float lireCapDeg() {
  int16_t mx, my, mz;
  if (!lireMagRaw(mx, my, mz)) return NAN;
  float x = (float)mx - MAG_OFFSET_X, y = (float)my - MAG_OFFSET_Y;
  float angle = atan2(y, x) * 180.0f / PI;
  if (angle < 0) angle += 360.0f;
  angle += MAG_ANGLE_CORRECTION;
  if (angle >= 360.0f) angle -= 360.0f;
  return angle;
}
void setHeadingReference() {
  const int N = 200; long smx=0, smy=0; int ok=0;
  for (int i=0; i<N; i++) {
    int16_t mx,my,mz;
    if (lireMagRaw(mx,my,mz)) { smx+=mx; smy+=my; ok++; }
    delay(5);
  }
  if (ok==0) return;
  float a = atan2((float)smy/ok,(float)smx/ok)*180.0f/PI;
  if (a<0) a+=360.0f;
  headingReferenceDeg = a;
}
void updateHeading() {
  int16_t mx,my,mz;
  if (!lireMagRaw(mx,my,mz)) return;
  float a = atan2((float)my,(float)mx)*180.0f/PI;
  if (a<0) a+=360.0f;
  headingDeg = a - headingReferenceDeg;
  if (headingDeg<0) headingDeg+=360.0f;
  if (headingDeg>=360.0f) headingDeg-=360.0f;
  float cap = lireCapDeg();
  if (!isnan(cap)) capDeg = cap;
}

// =====================================================
// MOTEURS INDIVIDUELS (séquences automatiques)
// pwm positif = avant, négatif = arrière
// =====================================================
void motorG(int pwm) {
  pwm = constrain(pwm,-255,255);
  if      (pwm>0) { ledcWrite(2,pwm); ledcWrite(3,0);    }
  else if (pwm<0) { ledcWrite(2,0);   ledcWrite(3,-pwm); }
  else            { ledcWrite(2,0);   ledcWrite(3,0);     }
}
void motorD(int pwm) {
  pwm = constrain(pwm,-255,255);
  if      (pwm>0) { ledcWrite(1,pwm); ledcWrite(0,0);    }
  else if (pwm<0) { ledcWrite(1,0);   ledcWrite(0,-pwm); }
  else            { ledcWrite(1,0);   ledcWrite(0,0);     }
}

void appliquerMouvement() {
  switch (etatActuel) {
    case AVANT:   avancer(vitesseCommande);       robotMoving=true;  break;
    case ARRIERE: reculer(vitesseCommande);        robotMoving=true;  break;
    case GAUCHE:  tournerGauche(vitesseCommande);  robotMoving=false; break;
    case DROITE:  tournerDroite(vitesseCommande);  robotMoving=false; break;
    case STOP:    stopMoteurs();                   robotMoving=false; break;
  }
}

// =====================================================
// UPDATE POSITION IMU
// =====================================================
void updatePosition() {
  unsigned long now = micros();
  if (lastUpdateMicros==0) { lastUpdateMicros=now; return; }
  float dt = (now-lastUpdateMicros)/1000000.0f;
  lastUpdateMicros = now;
  if (dt<=0.0f||dt>0.2f) return;

  int16_t rawX=read16(ADDR_IMU,OUTX_L_XL), rawY=read16(ADDR_IMU,OUTY_L_XL);
  float ax=rawX*0.000061f*9.81f-accelOffsetX, ay=rawY*0.000061f*9.81f-accelOffsetY;
  if (fabs(ax)<SEUIL_ACCEL) ax=0; if (fabs(ay)<SEUIL_ACCEL) ay=0;
  accelX_ms2=ax; accelY_ms2=ay;
  accelGlobale_ms2=sqrt(ax*ax+ay*ay);

  if (!robotMoving||etatActuel==STOP||etatActuel==GAUCHE||etatActuel==DROITE) {
    vitesseX_ms=vitesseY_ms=vitesseGlobale_ms=0; immobileCounter=0; return;
  }
  float theta=headingDeg*PI/180.0f;
  float axw=(ax*cos(theta)-ay*sin(theta))*GAIN_ACCEL_IMU;
  float ayw=(ax*sin(theta)+ay*cos(theta))*GAIN_ACCEL_IMU;
  vitesseX_ms+=axw*dt; vitesseY_ms+=ayw*dt;
  vitesseX_ms*=0.995f; vitesseY_ms*=0.995f;
  vitesseX_ms=constrain(vitesseX_ms,-1.5f,1.5f);
  vitesseY_ms=constrain(vitesseY_ms,-1.5f,1.5f);
  vitesseGlobale_ms=sqrt(vitesseX_ms*vitesseX_ms+vitesseY_ms*vitesseY_ms);
  posX_m+=vitesseX_ms*dt; posY_m+=vitesseY_ms*dt;
  addTracePoint(posX_m,posY_m);
}

// =====================================================
// ORIENTATION NORD
// =====================================================
void updateOrientationNord() {
  if (!orientationNordActive) return;
  float cap=lireCapDeg();
  if (isnan(cap)) { stopMoteurs(); orientationNordActive=false; return; }
  capDeg=cap;
  float err=cap; while(err>180)err-=360; while(err<-180)err+=360;
  err=-err;
  if (fabs(err)<=TOLERANCE_DEG) {
    stopMoteurs(); orientationNordActive=false;
    etatActuel=STOP; vitesseCommande=0; robotMoving=false; return;
  }
  int pwm=(fabs(err)<10)?95:(fabs(err)<20)?110:PWM_ROTATION;
  if (err>0){tournerGauche(pwm);etatActuel=GAUCHE;}
  else      {tournerDroite(pwm);etatActuel=DROITE;}
}

// =====================================================
// RESET GLOBAL
// =====================================================
void resetPositionEtVitesses() {
  posX_m=posY_m=0;
  accelX_ms2=accelY_ms2=accelGlobale_ms2=0;
  vitesseX_ms=vitesseY_ms=vitesseGlobale_ms=0;
  immobileCounter=0; robotMoving=false;
  clearTrace(); addTracePoint(0,0);
  resetEncodeurs(); resetOdometryEncodeurs();
  lastUpdateMicros=micros();
}

// =====================================================
// PRIMITIVE : AVANCER EN LIGNE DROITE
// =====================================================
void seqAvancerCm(float distCm) {
  resetEncodeurs();
  long ticksCible = cmToTicks(distCm);
  etatActuel = AVANT; robotMoving = true;
  long g0=0, d0=0;
  float segStart = distTotalCm;

  while (true) {
    server.handleClient();
    updateHeading(); updatePosition(); updateOdometryEncodeurs();
    long g,d; lireEncodeurs(g,d);
    distTotalCm = segStart + ticksToCm((g+d)/2);
    if ((g+d)/2 >= ticksCible - TOL_TICKS_DIST) {
      stopMoteurs(); etatActuel=STOP; robotMoving=false; delay(120); return;
    }
    float errCap = (float)((g-g0)-(d-d0));
    float corr   = constrain(GAIN_CORR_CAP*errCap,-40.f,40.f);
    g0=g; d0=d;
    motorG(constrain((int)(SEQ_PWM_AVANCE-corr),60,255));
    motorD(constrain((int)(SEQ_PWM_AVANCE+corr),60,255));
    delay(5);
  }
}

// =====================================================
// ROTATION EN 3 PHASES — 90° vers la droite
//
// PHASE A  (0° → 6°) : pivot pur
//   Roue droite = 0 (immobile)
//   Roue gauche = SEQ_PWM_ROT_G
//   Centre de rotation = axe roue droite
//   → Virage très serré, courbure maximale
//
// PHASE B  (6° → 90°) : accélération progressive roue droite
//   Roue gauche = SEQ_PWM_ROT_G (constante)
//   Roue droite = SEQ_PWM_ROT_G × progress  (0 → 1)
//   où progress = (angle - 45) / 45  (linéaire)
//   → Rayon de courbure augmente progressivement
//   → À 90° : les deux roues ont la même vitesse → ligne droite
//
// Angle mesuré par fusion gyro (70%) + encodeurs (30%)
// =====================================================
void seqRotationDiff90Droite() {
  resetEncodeurs();

  const float ANGLE_CIBLE    = 90.0f;
  const float ANGLE_PIVOT    = 10.0f / ENTRAXE_CM * (180.0f / PI); // phase A = 10 cm d'arc roue active

  float angleGyro   = 0.0f;
  float angleFusion = 0.0f;
  unsigned long tPrev = micros();

  etatActuel  = DROITE;
  robotMoving = false;

  Serial.println("  Rotation 3 phases...");

  while (true) {
    server.handleClient();
    updateHeading();

    // --- Temps écoulé ---
    unsigned long tNow = micros();
    float dt = (tNow - tPrev) / 1000000.0f;
    tPrev = tNow;
    if (dt <= 0 || dt > 0.1f) dt = 0.005f;

    // --- Gyro ---
    float gz = lireGyroZ_dps();
    angleGyro += gz * dt;

    // --- Encodeurs ---
    long g, d;
    lireEncodeurs(g, d);
    float arcG = ticksToCm(g);
    float arcD = ticksToCm(d);
    float angleEnc = ((arcG - arcD) / ENTRAXE_CM) * (180.0f / PI);

    // --- Fusion ---
    angleFusion = ALPHA_GYRO * angleGyro + (1.0f - ALPHA_GYRO) * angleEnc;

    // Mise à jour variable UI
    rotAngleCourant = angleFusion;

    // --- Condition d'arrêt ---
    if (angleFusion >= ANGLE_CIBLE - TOL_ANGLE_DEG) {
      stopMoteurs();
      etatActuel  = STOP;
      robotMoving = false;
      Serial.print("  Rotation terminee : ");
      Serial.print(angleFusion);
      Serial.println(" deg");
      delay(150);
      return;
    }

    // --- Calcul des PWM selon la phase ---
    int pwmG, pwmD;

    if (angleFusion < ANGLE_PIVOT) {
      // ── PHASE A : pivot pur (0° → 6°) ──────────────────
      // Roue droite immobile, roue gauche à vitesse constante
      pwmG = SEQ_PWM_ROT_G;
      pwmD = 0;

    } else {
      // ── PHASE B : accélération progressive (6° → 90°) ──
      // progress : 0.0 au début de la phase, 1.0 à 90°
      float progress = (angleFusion - ANGLE_PIVOT) / (ANGLE_CIBLE - ANGLE_PIVOT);
      progress = constrain(progress, 0.0f, 1.0f);

      // Roue gauche : légère correction si en retard
      float erreur = ANGLE_CIBLE - angleFusion;
      float corrG  = constrain(erreur * 0.4f, -12.0f, 12.0f);
      pwmG = constrain(SEQ_PWM_ROT_G + (int)corrG, 60, 180);

      // Roue droite : monte plus vite (expo < 1 = courbe concave)
      pwmD = (int)(pwmG * powf(progress, 0.7f));
      pwmD = constrain(pwmD, 0, 180);
    }

    motorG(pwmG);
    motorD(pwmD);
    delay(5);
  }
}

// =====================================================
// SEQUENCE ESCALIER
// 1. Avancer 20 cm (segment vertical)
// 2. Rotation différentielle 90° droite (stylo reste
//    dans la continuité verticale)
// 3. Avancer 10 cm (segment horizontal)
// =====================================================
// =====================================================
// ROTATION PIVOT 90° vers la GAUCHE
// Roue gauche = 0 (immobile, pivot)
// Roue droite = SEQ_PWM_ROT_G (tourne en avant)
// =====================================================
void seqRotationPivotGauche90() {
  resetEncodeurs();

  const float ANGLE_CIBLE   = 100.08f;
  const float ANGLE_PIVOT   = 10.0f / ENTRAXE_CM * (180.0f / PI); // phase A = 10 cm d'arc roue active
  // Tolérance réduite : on coupe les moteurs plus tard
  // pour compenser l'inertie qui fait s'arrêter le robot trop tôt
  const float TOL_LOCALE    = 0.3f;

  float angleGyro   = 0.0f;
  float angleFusion = 0.0f;
  unsigned long tPrev = micros();

  etatActuel  = GAUCHE;
  robotMoving = false;

  Serial.println("  Rotation 3 phases GAUCHE...");

  while (true) {
    server.handleClient();
    updateHeading();

    unsigned long tNow = micros();
    float dt = (tNow - tPrev) / 1000000.0f;
    tPrev = tNow;
    if (dt <= 0 || dt > 0.1f) dt = 0.005f;

    float gz = lireGyroZ_dps();
    angleGyro += (-gz) * dt;

    long g, d;
    lireEncodeurs(g, d);
    float arcG     = ticksToCm(g);
    float arcD     = ticksToCm(d);
    float angleEnc = ((arcD - arcG) / ENTRAXE_CM) * (180.0f / PI);

    angleFusion = ALPHA_GYRO * angleGyro + (1.0f - ALPHA_GYRO) * angleEnc;
    rotAngleCourant = angleFusion;

    // Arrêt avec tolérance locale réduite à 0.3°
    if (angleFusion >= ANGLE_CIBLE - TOL_LOCALE) {
      stopMoteurs();
      etatActuel  = STOP;
      robotMoving = false;
      Serial.print("  Rotation GAUCHE terminee : ");
      Serial.print(angleFusion);
      Serial.println(" deg");
      // Pas de delay ici : on laisse l'inertie finir les derniers degrés
      return;
    }

    int pwmG, pwmD;

    if (angleFusion < ANGLE_PIVOT) {
      pwmD = SEQ_PWM_ROT_G;
      pwmG = 0;
    } else {
      float progress = (angleFusion - ANGLE_PIVOT) / (ANGLE_CIBLE - ANGLE_PIVOT);
      progress = constrain(progress, 0.0f, 1.0f);

      float erreur = ANGLE_CIBLE - angleFusion;
      float corrD  = constrain(erreur * 0.4f, -12.0f, 12.0f);
      pwmD = constrain(SEQ_PWM_ROT_G + (int)corrD, 60, 180);

      // Roue gauche : monte plus lentement (expo > 1 = courbe convexe)
      pwmG = (int)(pwmD * powf(progress, 1.4f));
      pwmG = constrain(pwmG, 0, 180);
    }

    motorG(pwmG);
    motorD(pwmD);
    delay(5);
  }
}

// =====================================================
// SEQUENCE ESCALIER COMPLET
//
// Le virage droite est un pivot pur (roue D = 0).
// L'arc tracé par le stylo pendant ce pivot vaut :
//   arc = STYLO_AVANT_CM × (π/2) ≈ 13 × 1.571 ≈ 20.4 cm
// Mais ce déplacement est dans l'axe vertical du robot
// (continuation du trait), donc il "consomme" de la distance.
//
// On veut 10 cm utiles APRES le virage droite.
// Le virage lui-même ne déplace pas le robot latéralement
// mais avance le stylo de STYLO_AVANT_CM vers la droite.
// Les 10 cm après sont donc bien 10 cm horizontaux nets.
//
// Séquence :
// 1. Avancer 20 cm  (segment vertical)
// 2. Virage 90° droite  (pivot roue D)
// 3. Avancer 10 cm  (segment horizontal droite)
// 4. Virage 90° gauche  (pivot roue G)
// 5. Avancer 10 cm  (segment vertical bas)
// =====================================================
void executerSequenceEscalier() {
  Serial.println("=== SEQUENCE ESCALIER ===");
  distTotalCm = 0.0f;

  Serial.print("Segment 1 : avance "); Serial.print(distSeg1Cm); Serial.println(" cm");
  seqAvancerCm(distSeg1Cm);
  delay(200);

  Serial.println("Virage 1 : 90 deg droite");
  seqRotationDiff90Droite();
  delay(200);

  Serial.print("Segment 2 : avance "); Serial.print(distSeg2Cm); Serial.println(" cm");
  seqAvancerCm(distSeg2Cm);
  delay(200);

  Serial.println("Virage 2 : 90 deg gauche");
  seqRotationPivotGauche90();
  delay(200);

  Serial.print("Segment 3 : avance "); Serial.print(distSeg3Cm); Serial.println(" cm");
  seqAvancerCm(distSeg3Cm);
  delay(200);

  sequenceActive = false;
  Serial.println("=== FIN SEQUENCE ESCALIER ===");
}

}

void setup(){
  Serial.begin(9600);

  pinMode(EN_D,OUTPUT); pinMode(EN_G,OUTPUT);
  digitalWrite(EN_D,HIGH); digitalWrite(EN_G,HIGH);

  ledcSetup(0,1000,8); ledcSetup(1,1000,8);
  ledcSetup(2,1000,8); ledcSetup(3,1000,8);
  ledcAttachPin(IN_1_D,0); ledcAttachPin(IN_2_D,1);
  ledcAttachPin(IN_1_G,2); ledcAttachPin(IN_2_G,3);

  pinMode(ENC_G_CH_A,INPUT_PULLUP); pinMode(ENC_G_CH_B,INPUT_PULLUP);
  pinMode(ENC_D_CH_A,INPUT_PULLUP); pinMode(ENC_D_CH_B,INPUT_PULLUP);
  attachInterrupt(digitalPinToInterrupt(ENC_G_CH_A),isrEncG,CHANGE);
  attachInterrupt(digitalPinToInterrupt(ENC_D_CH_A),isrEncD,CHANGE);

  stopMoteurs();
  Wire.begin(SDA_PIN,SCL_PIN);
  initIMU(); initMagnetometre();
  delay(500);

  calibrerAccel();
  calibrerGyro();        // ← nouveau : calibration offset gyroscope Z
  setHeadingReference();
  resetPositionEtVitesses();

  WiFi.softAP(ssid,password);
  Serial.print("Adresse : http://"); Serial.println(WiFi.softAPIP());

  server.on("/",                  handleRoot);
  server.on("/data",              handleData);
  server.on("/reset",             handleReset);
  server.on("/cleartrace",        handleClearTrace);
  server.on("/nord",              handleNord);
  server.on("/avant",             handleAvant);
  server.on("/arriere",           handleArriere);
  server.on("/gauche",            handleGauche);
  server.on("/droite",            handleDroite);

  server.on("/recalibrer",        handleRecalibrerAccel);
  server.on("/seqparams",         handleSeqParams);
  server.on("/sequenceEscalier",  handleSequenceEscalier);

  server.begin();
  Serial.println("Pret");
  lastUpdateMicros=micros();
}
